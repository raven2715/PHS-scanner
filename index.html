⚡ Python SDS Comparison Script

import re
from pdfminer.high_level import extract_text

def extract_sds_section_3(sds_text: str) -> str:
    """Isolates Section 3 (Composition) from a standard 16-section SDS."""
    pattern = re.compile(
        r"(?:3\.\s*COMPOSITION|Composition/information on ingredients|SECTION\s*3)"
        r"(.*?)"
        r"(?:4\.\s*FIRST|SECTION\s*4|First-aid measures)",
        re.IGNORECASE | re.DOTALL
    )
    match = pattern.search(sds_text)
    return match.group(1).strip() if match else sds_text

def parse_phs_list(phs_pdf_path: str) -> set:
    """Extracts cleaned chemical names from the CDC PHS List PDF."""
    raw_text = extract_text(phs_pdf_path)
    lines = [line.strip() for line in raw_text.splitlines() if line.strip()]
    
    exclude_keywords = {
        'chemical hygiene plan', 'plan-000001-csa', 'revision no:', 'effective date:', 
        'page', 'table 7:', 'chemicals a-hyd', 'chemicals i-z', 'select toxins', 
        'sodium azide is', 'extremely toxic', 'amount can be', 'powder. upon', 
        'sodium azide reacts', 'which are shock-sensitive', 'it can also', 
        'store on metal', 'sodium azide and', 'safety cabinet', 'include safety', 
        'as double nitrile', 'workers and visitors', 'toxins at any', 
        'assessment approval', 'to select toxins', 'responsible for', 
        'location has been', 'to specific agents', 'application.'
    }
    
    phs_chemicals = set()
    for line in lines:
        lower_line = line.lower()
        skip = False
        for kw in exclude_keywords:
            if kw in lower_line:
                skip = True
                break
        if skip:
            continue
        cleaned = line.strip(" ,.-•\t")
        if len(cleaned) > 2 and "highly explosive" not in cleaned and "biological" not in cleaned:
            phs_chemicals.add(cleaned)
            
    return phs_chemicals

def compare_sds_to_reference(sds_path: str, ref_path: str) -> dict:
    sds_full_text = extract_text(sds_path)
    sds_sec_3 = extract_sds_section_3(sds_full_text)
    phs_chemicals = parse_phs_list(ref_path)
    
    exact_matches = []
    for chem in phs_chemicals:
        std_chem = chem.lower()
        if std_chem == "trireagent":
            std_chem = "tri reagent"
            
        if std_chem in sds_sec_3.lower() or (std_chem in sds_full_text.lower() and std_chem == "tri reagent"):
            exact_matches.append(chem)
            
    return {
        "match_found": len(exact_matches) > 0,
        "matches": exact_matches
    }

if __name__ == "__main__":
    results = compare_sds_to_reference("RNAzol_PHS.pdf", "PHS List pages.pdf")
    print("Match Status:", results['match_found'])
    print("Matched PHS Items:", results['matches'])
