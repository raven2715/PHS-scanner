  function escapeRegExp(str) {

    return str.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');

  }

 

  function extractSections3(fullText) {

    // Isolate all Section 3 blocks up to Section 4 (supports multi-kit SDS PDFs)

    const primaryRegex = /(?:section\s*0?3\b|3\s+Composition\/information\s+on\s+ingredients)([\s\S]*?)(?=section\s*0?4\b|4\s+First-aid\s+measures)/gi;

    const matches = [];

    let match;

    while ((match = primaryRegex.exec(fullText)) !== null) {

      matches.push(match[1].trim());

    }

    return matches.length > 0 ? matches : [fullText]; // Fallback to full text if non-standard headers

  }

 

  async function scanSDS() {

    const fileInput = document.getElementById('sdsFileInput');

    const statusText = document.getElementById('statusText');

    const resultContainer = document.getElementById('resultContainer');

    resultContainer.style.display = 'none';

 

    if (!fileInput.files.length) {

      alert("Please select an SDS PDF file first.");

      return;

    }

 

    statusText.textContent = "⏳ Extracting text and scanning Section 3...";

    const file = fileInput.files[0];

 

    try {

      const arrayBuffer = await file.arrayBuffer();

      const pdf = await pdfjsLib.getDocument({ data: arrayBuffer }).promise;

      let fullText = "";

 

      for (let i = 1; i <= pdf.numPages; i++) {

        const page = await pdf.getPage(i);

        const textContent = await page.getTextContent();

        const pageText = textContent.items.map(item => item.str).join(" ");

        fullText += pageText + "\n";

      }

 

      const section3Parts = extractSections3(fullText);

      let totalMatches = 0;

      let htmlOutput = "";

 

      section3Parts.forEach((secText, idx) => {

        const partMatches = [];

 

        PHS_CHEMICALS.forEach(chem => {

          const subNames = [chem];

          // Split parenthetical synonyms (e.g., "Dichloromethane (methylene chloride)")

          if (chem.includes("(")) {

            const parenMatches = [...chem.matchAll(/\((.*?)\)/g)];

            parenMatches.forEach(m => {

              if (m[1].length >= 3 && !m[1].includes("comp.") && !m[1].includes("solid")) {

                subNames.push(m[1]);

              }

            });

            const preParen = chem.split("(")[0].trim();

            if (preParen.length >= 3) subNames.push(preParen);

          }

 

          for (const sn of subNames) {

            const regex = new RegExp(`\\b${escapeRegExp(sn)}\\b`, "i");

            if (regex.test(secText)) {

              partMatches.push(`<strong>${chem}</strong> (matched via <em>"${sn}"</em>)`);

              totalMatches++;

              break;

            }

          }

        });

 

        if (section3Parts.length > 1) {

          htmlOutput += `<div class="part-header">Component / Sub-SDS Part ${idx + 1}:</div>`;

        }

        if (partMatches.length > 0) {

          htmlOutput += `<ul>${partMatches.map(m => `<li>⚠️ ${m}</li>`).join("")}</ul>`;

        } else if (section3Parts.length > 1) {

          htmlOutput += `<div>✅ No PHS matches in this component.</div>`;

        }

      });

 

      statusText.textContent = "";

      resultContainer.style.display = 'block';

 

      if (totalMatches > 0) {

        resultContainer.className = "result-box match-alert";

        resultContainer.innerHTML = `<strong>🚨 PHS MATCH DETECTED (${totalMatches} found)</strong><br>${htmlOutput}`;

      } else {

        resultContainer.className = "result-box clear-alert";

        resultContainer.innerHTML = `<strong>✅ No PHS Matches Found</strong><br>Scanned ${section3Parts.length} Section 3 composition block(s) against 137 PHS chemicals.`;

      }

    } catch (err) {

      statusText.textContent = "❌ Error reading PDF: " + err.message;

    }

  }
